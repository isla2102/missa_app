function getJuDate(type, value) {
	$.ajax({
		type: 'POST',
		dataType: 'JSON',
		url: "ajax_get_date.asp",
		data: type+"="+value,
		success: function (data) {
			var options_str = "<option value=''>전례주일 별</option>"
			$.each(data, function(i, value) {
				options_str += "<option value='" + value.ju_date + "'>" + value.ju_title + "</option>";
			});
			$("select[name=s_date]").empty().append(options_str);	
		}, //success
		error:function(request,status,error){
			alert("code:"+request.status+"\n"+"message:"+request.responseText+"\n"+"error:"+error);
		}
	});
	//주보메인검색은 캘린더날짜/년도 택 1 선택 시 해당 전례주일이 나오는 로직
	if(type=="s_date") { //캘린더에서 날짜 선택 시
		$("select[name=s_year]").val("");
	}else { //년도 선택 시
		$(".datepicker").val("");
	}
}
