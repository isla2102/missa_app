    function goPrayer(menu,pgubun,ingid,sgubun){
        $.ajax({
            type: "GET", //"POST",
            url: "getPrayer.asp",
            data: {menu:menu, pgubun:pgubun,ingId:ingid,sgubun:sgubun},

            dataType: "html",
            error: function (msg) {
                //alert('error');
            },
            beforeSend: function() {
                //마우스 커서를 로딩 중 커서로 변경
                $('html').css("cursor", "wait");
            },
            complete: function() {
                //마우스 커서를 원래대로 돌린다
                $('html').css("cursor", "auto");
            },

            success: function (msg) {
                //alert(msg);
                $("#prayerContent").html(msg); //innerHTML 
                $( 'html, body' ).animate( { scrollTop : 0 }, 400 );
            }
        });
    }
